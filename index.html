export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; // Permite as requisições OPTIONS if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: corsHeaders }); } const url = new URL(request.url); // ===================================================== // SITE // ===================================================== // Quando abrir o endereço principal, entrega o index.html if ( (request.method === "GET" || request.method === "HEAD") && (url.pathname === "/" || !url.pathname.startsWith("/api")) ) { if (env.ASSETS) { return env.ASSETS.fetch(request); } return new Response( "Erro: o armazenamento de arquivos (ASSETS) não está configurado.", { status: 500, headers: { "Content-Type": "text/plain; charset=UTF-8", ...corsHeaders } } ); } // ===================================================== // API DO RAPHAEL TENSEI // ===================================================== if (url.pathname === "/api" || url.pathname === "/api/chat") { if (request.method === "GET") { return new Response( JSON.stringify({ reply: "Raphael Tensei online." }), { status: 200, headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } if (request.method === "POST") { try { const data = await request.json(); const pergunta = data.message || data.question || data.prompt || ""; if (!pergunta.trim()) { return new Response( JSON.stringify({ reply: "Digite uma pergunta." }), { status: 400, headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } // Resposta básica do Raphael Tensei const resposta = "Olá. Eu sou Raphael Tensei. Recebi sua mensagem: " + pergunta; return new Response( JSON.stringify({ reply: resposta }), { status: 200, headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } catch (error) { return new Response( JSON.stringify({ reply: "Não consegui processar sua mensagem." }), { status: 400, headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } } } // ===================================================== // ROTA NÃO ENCONTRADA // ===================================================== return new Response( JSON.stringify({ error: "Rota não encontrada." }), { status: 404, headers: { "Content-Type": "application/json; charset=UTF-8", ...corsHeaders } } ); } };
